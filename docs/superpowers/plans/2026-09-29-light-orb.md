# Light Orb Shader Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A single-file Three.js demo that reproduces `animation_02.mp4` 1:1: a soft glowing orb (blue/amber lobes) with a closed, saddle-shaped ribbon of light inside, looping seamlessly every 5 s.

**Architecture:** One `index.html` with an importmap (Three.js r170 + lil-gui 0.19.2 from jsDelivr). Three additive layers (sphere shell, ribbon, ribbon halo), each registering a per-frame `layerUpdates` hook that receives the loop phase. Post chain: `RenderPass → UnrealBloomPass → 2×(blur H + blur V) → OutputPass (ACES, sRGB) → finish`. A `?capture=WxH` mode plus `window.__orb` makes frames deterministic so a Python script can score them against frames extracted from the video.

**Tech Stack:** Three.js 0.170.0, lil-gui 0.19.2, GLSL (WebGL 2), Python 3 + numpy + Pillow (verification only), ffmpeg (frame extraction).

Spec: `docs/superpowers/specs/2026-09-29-light-orb-design.md`

```python
#!/usr/bin/env python3
"""Score rendered orb frames against frames from the reference video.

Usage: compare.py REF_DIR RENDER_DIR [--sheet OUT.png]

Reads REF_DIR/ref_t{0..5}.png and RENDER_DIR/render_t{0..5}.png, prints one
PASS/FAIL line per acceptance check from the design spec, and exits 0 only
when every check passes.
"""
import argparse
import math
import sys
from pathlib import Path

import numpy as np
from PIL import Image, ImageDraw

TIMES = [0, 1, 2, 3, 4, 5]
DIAMETER_TOL = 0.05   # +-5 % of the reference diameter
COLOR_TOL = 15.0      # per channel, 0..255
LOOP_TOL = 0.5        # mean abs diff between t=0 and t=5, 0..255
CELL = 240            # contact sheet cell size


def load(path):
    return np.asarray(Image.open(path).convert('RGB'), dtype=np.float32)


def diameter(img):
    """Largest bbox side of pixels brighter than 12/255 (orb incl. glow)."""
    ys, xs = np.nonzero(img.max(axis=2) > 12)
    if len(xs) == 0:
        return 0.0
    return float(max(xs.max() - xs.min(), ys.max() - ys.min()))


def color_means(frames):
    """Mean colour of 'blue' and 'amber' pixels over a set of frames."""
    px = np.concatenate([f.reshape(-1, 3) for f in frames])
    lum = px.sum(axis=1)
    blue = px[(px[:, 2] > px[:, 0] + 30) & (lum > 150)]
    amber = px[(px[:, 0] > px[:, 2] + 60) & (lum > 200)]

    def mean(a):
        return a.mean(axis=0) if len(a) else np.full(3, np.nan)

    return mean(blue), mean(amber)


def corners_black(img, k=20):
    h, w, _ = img.shape
    patches = [img[:k, :k], img[:k, w - k:], img[h - k:, :k], img[h - k:, w - k:]]
    return max(float(p.max()) for p in patches) == 0.0


def hexcol(c):
    if np.any(np.isnan(c)):
        return 'none'
    return '#%02x%02x%02x' % tuple(int(round(v)) for v in c)


def centre_crop(img):
    h, w, _ = img.shape
    s = int(min(h, w) * 0.6)
    y0, x0 = (h - s) // 2, (w - s) // 2
    crop = Image.fromarray(img[y0:y0 + s, x0:x0 + s].astype(np.uint8))
    return crop.resize((CELL, CELL), Image.LANCZOS)


def contact_sheet(refs, renders, out):
    """Top row: video, bottom row: render, one column per timestamp."""
    sheet = Image.new('RGB', (CELL * len(TIMES), CELL * 2), 'black')
    draw = ImageDraw.Draw(sheet)
    for i, t in enumerate(TIMES):
        sheet.paste(centre_crop(refs[i]), (i * CELL, 0))
        sheet.paste(centre_crop(renders[i]), (i * CELL, CELL))
        draw.text((i * CELL + 6, 6), f'video t={t}s', fill=(190, 190, 190))
        draw.text((i * CELL + 6, CELL + 6), f'render t={t}s', fill=(190, 190, 190))
    sheet.save(out)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('ref_dir')
    ap.add_argument('render_dir')
    ap.add_argument('--sheet')
    args = ap.parse_args()

    refs = [load(Path(args.ref_dir) / f'ref_t{t}.png') for t in TIMES]
    renders = [load(Path(args.render_dir) / f'render_t{t}.png') for t in TIMES]

    results = []

    def check(name, ok, detail=''):
        results.append(bool(ok))
        print(f"{'PASS' if ok else 'FAIL'}  {name}: {detail}")

    same_size = all(r.shape == refs[0].shape for r in renders)
    check('frame size', same_size,
          f'{renders[0].shape[1]}x{renders[0].shape[0]} vs {refs[0].shape[1]}x{refs[0].shape[0]}')
    if not same_size:
        sys.exit(1)

    ref_d = float(np.mean([diameter(f) for f in refs]))
    for t, img in zip(TIMES, renders):
        d = diameter(img)
        check(f'diameter t={t}', abs(d - ref_d) <= DIAMETER_TOL * ref_d,
              f'{d:.0f}px vs ref {ref_d:.0f}px')

    ref_blue, ref_amber = color_means(refs)
    ren_blue, ren_amber = color_means(renders)
    for name, ref_c, ren_c in (('blue mean', ref_blue, ren_blue), ('amber mean', ref_amber, ren_amber)):
        diff = np.abs(ren_c - ref_c)
        worst = float(np.max(diff)) if not np.any(np.isnan(diff)) else math.inf
        check(name, worst <= COLOR_TOL,
              f'render {hexcol(ren_c)} vs ref {hexcol(ref_c)} (worst channel diff {worst:.1f})')

    seam = float(np.abs(renders[0] - renders[-1]).mean())
    check('loop seam t=0 vs t=5', seam < LOOP_TOL, f'{seam:.3f}/255')
    check('corners black', all(corners_black(r) for r in renders))

    for t, a, b in zip(TIMES, refs, renders):
        print(f'info  MAE vs video t={t}: {np.abs(a - b).mean():.2f}/255')

    if args.sheet:
        contact_sheet(refs, renders, args.sheet)
        print(f'info  sheet written to {args.sheet}')

    sys.exit(0 if all(results) else 1)


if __name__ == '__main__':
    main()
```

```python
#!/usr/bin/env python3
"""Static server for the orb page that also saves captured frames.

Usage: capture_server.py ROOT OUT_DIR PORT
GET  /...               -> files from ROOT (no caching)
POST /save?name=<file>  -> request body written to OUT_DIR/<basename(file)>
"""
import http.server
import os
import sys
import urllib.parse

ROOT, OUT_DIR, PORT = sys.argv[1], sys.argv[2], int(sys.argv[3])
os.makedirs(OUT_DIR, exist_ok=True)


class Handler(http.server.SimpleHTTPRequestHandler):
    def __init__(self, *args, **kwargs):
        super().__init__(*args, directory=ROOT, **kwargs)

    def end_headers(self):
        self.send_header('Cache-Control', 'no-store')
        super().end_headers()

    def do_POST(self):
        url = urllib.parse.urlparse(self.path)
        names = urllib.parse.parse_qs(url.query).get('name')
        if url.path != '/save' or not names:
            self.send_error(404)
            return
        body = self.rfile.read(int(self.headers.get('Content-Length', 0)))
        with open(os.path.join(OUT_DIR, os.path.basename(names[0])), 'wb') as f:
            f.write(body)
        self.send_response(204)
        self.end_headers()


http.server.ThreadingHTTPServer(('127.0.0.1', PORT), Handler).serve_forever()
```

```html
// and wobble inside the sphere

// ---------------------------------------------------------------------------

const bloom = new UnrealBloomPass(new THREE.Vector2(1, 1), params.bloomStrength, params.bloomRadius, params.bloomThreshold);
composer.addPass(bloom);

const fullscreenVert = /* glsl */`
  varying vec2 vUv;
  void main() { vUv = uv; gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0); }
`;

// separable gaussian blur, run in linear HDR space before tone mapping
const blurShader = {
  uniforms: {
    tDiffuse: { value: null },
    uDir: { value: new THREE.Vector2(1, 0) },
    uRadius: { value: 0 },
    uResolution: { value: new THREE.Vector2(1, 1) },
  },
  vertexShader: fullscreenVert,
  fragmentShader: /* glsl */`
    uniform sampler2D tDiffuse;
    uniform vec2 uDir;
    uniform float uRadius;
    uniform vec2 uResolution;
    varying vec2 vUv;
    void main() {
      // 13 taps spread over the radius, gaussian weights (sigma = radius / 2)
      vec2 stepUv = uDir * (uRadius / 6.0) / uResolution;
      vec4 sum = vec4(0.0);
      float wsum = 0.0;
      for (int i = -6; i <= 6; i++) {
        float x = float(i);
        float w = exp(-(x * x) / 18.0);
        sum += texture2D(tDiffuse, vUv + stepUv * x) * w;
        wsum += w;
      }
      gl_FragColor = sum / wsum;
    }
  `,
};
// two H+V rounds: the second one only runs for large radii, where 13 taps
// over one round would start to show banding
const blurPasses = [0, 1, 2, 3].map((i) => {
  const p = new ShaderPass(blurShader);
  if (i % 2 === 1) p.uniforms.uDir.value.set(0, 1);
  composer.addPass(p);
  return p;
});

composer.addPass(new OutputPass());

// mapping (grain in linear space lifts the blacks a lot)

  uniforms: {
    tDiffuse: { value: null },
    uChromatic: { value: 0 },
    uGrain: { value: 0 },
    uVignette: { value: 0 },
    uTime: { value: 0 },
    uAspect: { value: 1 },

  vertexShader: fullscreenVert,
  fragmentShader: /* glsl */`
    uniform sampler2D tDiffuse;
    uniform float uChromatic;
    uniform float uGrain;
```

```js
// ---------------------------------------------------------------------------
// Sphere shell: a soft glass of light. Brightness comes from a fresnel term
// (bright limb, dark middle); the far wall uses a softer falloff so the lower
// half reads as a lit bowl. Colour comes from two opposite lobes (cool /
// warm) along a tilted direction that circles the vertical axis once per
// loop, so the colours swap hemispheres.
// ---------------------------------------------------------------------------
const sphereVert = /* glsl */`
  varying vec3 vNormalV;
  varying vec3 vViewV;
  varying vec3 vNormalO;

  void main() {
    vec4 mv = modelViewMatrix * vec4(position, 1.0);
    vNormalV = normalize(normalMatrix * normal);
    vViewV = normalize(-mv.xyz);
    vNormalO = normal;
    gl_Position = projectionMatrix * mv;
  }
`;

const sphereFrag = /* glsl */`
  uniform vec3 uCool;
  uniform vec3 uWarm;
  uniform vec3 uLobeDir;
  uniform float uFresnelPower;
  uniform float uShell;
  uniform float uInterior;
  uniform float uTop;
  uniform float uLobeSoft;

  varying vec3 vNormalV;
  varying vec3 vViewV;
  varying vec3 vNormalO;

  void main() {
    float facing = abs(dot(normalize(vNormalV), normalize(vViewV)));
    float fres = gl_FrontFacing
      ? pow(1.0 - facing, uFresnelPower) * uShell
      : pow(1.0 - facing, uFresnelPower * 0.5) * uInterior;

    vec3 n = normalize(vNormalO);
    vec3 col = mix(uCool, uWarm, smoothstep(-uLobeSoft, uLobeSoft, dot(n, uLobeDir)));
    // the upper cap is darker than the lower bowl
    col *= mix(uTop, 1.0, smoothstep(0.6, -0.6, n.y));
    gl_FragColor = vec4(col * fres, 1.0);
  }
`;
```

```js
// ---------------------------------------------------------------------------
// Ribbon: a dense plane whose vertices are wrapped in the vertex shader into
// a closed saddle-shaped loop (a warped ring) inside the sphere:
//   centre(th) = R * (cos th, h * sin(2 th + phi), sin th)
// The band runs mostly vertically and rolls around its tangent by
// twist * sin(th + phi), so its apparent width changes around the loop.
// phi advances once per loop, so the crest of the saddle travels around.
// ---------------------------------------------------------------------------
const ribbonVert = /* glsl */`
  uniform float uLoopR;
  uniform float uSaddleH;
  uniform float uWidth;
  uniform float uTwist;
  uniform float uPhi;

  varying vec3 vNormalV;
  varying vec3 vViewV;
  varying float vV;
  varying float vFront;

  vec3 ribbonPos(float u, float v) {
    float th = 6.2831853 * u;
    float s = 2.0 * th + uPhi;
    vec3 centre = uLoopR * vec3(cos(th), uSaddleH * sin(s), sin(th));
    vec3 tangent = normalize(vec3(-sin(th), 2.0 * uSaddleH * cos(s), cos(th)));
    vec3 radial = vec3(cos(th), 0.0, sin(th));
    vec3 up = normalize(cross(tangent, radial));
    float roll = uTwist * sin(th + uPhi);
    vec3 across = cos(roll) * up + sin(roll) * radial;
    return centre + across * (v * uWidth);
  }

  void main() {
    float u = uv.x;
    float v = uv.y * 2.0 - 1.0;
    vec3 p = ribbonPos(u, v);

    float e = 0.001;
    vec3 du = ribbonPos(u + e, v) - ribbonPos(u - e, v);
    vec3 dv = ribbonPos(u, v + e) - ribbonPos(u, v - e);
    // guard against a zero-length normal (NaN would be smeared by the bloom)
    vec3 c = cross(du, dv);
    float cl = length(c);
    vec3 n = cl > 1e-9 ? c / cl : vec3(0.0, 0.0, 1.0);

    vec4 mv = modelViewMatrix * vec4(p, 1.0);
    float originZ = (modelViewMatrix * vec4(0.0, 0.0, 0.0, 1.0)).z;
    vNormalV = normalize(normalMatrix * n);
    vViewV = normalize(-mv.xyz);
    vV = v;
    // -1 at the back of the loop, +1 at the front
    vFront = clamp((mv.z - originZ) / max(uLoopR, 1e-4), -1.0, 1.0);
    gl_Position = projectionMatrix * mv;
  }
`;

const ribbonFrag = /* glsl */`
  uniform vec3 uBodyColor;
  uniform float uBodyIntensity;
  uniform float uFoldPower;
  uniform float uFoldIntensity;
  uniform float uRimWidth;
  uniform float uRimIntensity;
  uniform vec3 uRimColorA;
  uniform vec3 uRimColorB;
  uniform float uBackDim;
  uniform float uGain;

  varying vec3 vNormalV;
  varying vec3 vViewV;
  varying float vV;
  varying float vFront;

  void main() {
    float nl = length(vNormalV);
    float facing = nl > 1e-5 ? abs(dot(vNormalV / nl, normalize(vViewV))) : 0.0;

  #ifdef SOFT_CORE
    // halo: a wide, soft gaussian across the band that spills into the shell
    vec3 col = uBodyColor * exp(-vV * vV * 2.5);
  #else
    // translucent amber body, brighter where the band faces the camera
    vec3 col = uBodyColor * uBodyIntensity * (0.25 + 0.75 * facing);

    // cream fold where the band turns edge-on to the camera
    col += vec3(1.0, 0.93, 0.82) * pow(1.0 - facing, uFoldPower) * uFoldIntensity;

    // rims on both edges: cream outermost, fading to amber inside
    float s = (1.0 - abs(vV)) / uRimWidth;
    vec3 rimCol = mix(uRimColorA, uRimColorB, smoothstep(0.5, 2.0, s));
    col += rimCol * exp(-s * s * 0.18) * uRimIntensity;
  #endif

    // the far side of the loop is fainter
    col *= mix(uBackDim, 1.0, smoothstep(-1.0, 1.0, vFront)) * uGain;
    if (any(isnan(col)) || any(isinf(col))) col = vec3(0.0);
    gl_FragColor = vec4(clamp(col, 0.0, 64.0), 1.0);
  }
`;
```

```js
const lipVert = /* glsl */`
  ${cutGlsl}
  uniform float uRadius;
  uniform float uWidth;
  uniform float uCurl;

  varying vec3 vNormalV;
  varying vec3 vViewV;
  varying float vV;
  varying float vFront;

  vec3 lipPos(float u, float v) {
    float th = 6.2831853 * u;
    float y = clamp(cutY(th), -0.97, 0.97);
    float r = sqrt(1.0 - y * y);
    vec3 onSphere = vec3(cos(th) * r, y, sin(th) * r);
```

```markdown
# Light Orb Shader

A soft orb of light with a saddle-shaped ribbon glowing inside it, rendered in real time in the browser with [Three.js](https://threejs.org). A companion to [Light Ribbon Shader](https://github.com/marcinkwiatkowski605/light-ribbon-shader).

![Light Orb preview](preview.jpg)

The orb is made of two translucent shells of light, each painted in cool blue and warm amber, that meet along a tilted, wavy line and overlap slightly, like a sphere cut in half. The edge of the upper shell catches the light and reads as a glowing ribbon. As the cut turns, the colours and the ribbon travel around the orb. The animation loops seamlessly every 5 seconds.

Every value can be changed live from the control panel.

## Run it locally

It is a single HTML file with no build step. Open `index.html` in a modern browser with WebGL 2. If the browser blocks it when opened from disk, start a local server in this folder (`npx serve .` or `python3 -m http.server`) and open the address it prints.

Three.js and lil-gui load from the jsDelivr CDN, so the first load needs an internet connection.

## Controls

The panel in the top-right corner has four groups:

| Group | What it changes |
| --- | --- |
| **Shape** | Sphere radius, view tilt, the cut between the shells (tilt, saddle height, offset), shell overlap and edge softness, cut and saddle cycles per loop and their phases, loop length, orb size |
| **Shells** | Fresnel power, shell and interior brightness, limb fade, cool and warm colours, colour softness and phase, edge glow and its width |
| **Lip** | Band width and curl, body colour and brightness, fold sharpness and brightness, rim width, brightness and colours, back side brightness, halo brightness and width |
| **Post-processing** | Bloom strength, radius and threshold, blur, exposure, grain, chromatic aberration, vignette |
| **Pause** | Freezes the animation |

All motion is set in whole cycles per loop, so the animation always loops without a seam.

## How it works

1. **The cut.** The line where the shells meet is a height on the sphere for each azimuth θ: `cutTilt·sin(θ − ψ) + saddleHeight·sin(2θ + φ) + cutOffset`. The first term tilts the split, the second bends it into a saddle. `ψ` and `φ` advance by whole cycles per loop. The same GLSL function is shared by the shell and lip shaders.
2. **Two shells and a lip.** One sphere mesh is drawn twice with additive blending: once keeping everything above the cut (plus a small overlap), once everything below. Each shell is lit by a fresnel term that fades out right at the silhouette, carries two colours around its azimuth (reversed on the lower shell) and glows along its own edge. A thin band (256 × 24 segments) built in the vertex shader runs exactly along the cut on the sphere surface: a translucent amber body, a cream fold where it turns edge-on, and rims along both edges, plus a soft halo copy.
3. **Post-processing chain.** `RenderPass → UnrealBloomPass → separable Gaussian blur (one or two rounds) → OutputPass (ACES tone mapping, sRGB) → final pass (chromatic aberration, vignette, grain)`. The blur radius is a fraction of the screen height, so the orb looks equally soft on a phone and on a 4K screen.

Add `?capture=1440x1080` to the URL to render fixed-size frames with no panel and no animation loop; `window.__orb.setTime(t)` and `window.__orb.renderFrame()` then produce any frame on demand.

- [lil-gui](https://github.com/georgealways/lil-gui) 0.19 for the control panel
- Custom GLSL vertex and fragment shaders
```
